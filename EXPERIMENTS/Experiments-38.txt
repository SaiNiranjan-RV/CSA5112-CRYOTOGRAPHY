
#include <stdio.h>

int mod(int x) {
    return (x % 26 + 26) % 26;
}

int inverse(int x) {
    for (int i = 1; i < 26; i++) {
        if ((x * i) % 26 == 1)
            return i;
    }
    return -1;
}

int main() {
    int P[2][2], C[2][2];
    int det, invDet;
    int Pinv[2][2], K[2][2];

    printf("Enter plaintext matrix (2x2):\n");

    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            scanf("%d", &P[i][j]);

    printf("Enter ciphertext matrix (2x2):\n");

    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            scanf("%d", &C[i][j]);

    det = mod(P[0][0] * P[1][1]
            - P[0][1] * P[1][0]);

    invDet = inverse(det);

    if (invDet == -1) {
        printf("Plaintext matrix is not invertible!\n");
        return 0;
    }

    Pinv[0][0] = mod(invDet * P[1][1]);
    Pinv[0][1] = mod(-invDet * P[0][1]);
    Pinv[1][0] = mod(-invDet * P[1][0]);
    Pinv[1][1] = mod(invDet * P[0][0]);

    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            K[i][j] = 0;

            for (int k = 0; k < 2; k++)
                K[i][j] += C[i][k] * Pinv[k][j];

            K[i][j] = mod(K[i][j]);
        }
    }

    printf("\nRecovered Hill Cipher Key:\n");

    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++)
            printf("%d ", K[i][j]);

        printf("\n");
    }

    return 0;
}
