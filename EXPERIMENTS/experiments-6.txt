
#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main()
{
    char text[100], key[100];
    int i, j = 0;

    printf("Enter the plaintext: ");
    fgets(text, sizeof(text), stdin);

    printf("Enter the key: ");
    scanf("%99s", key);

    int keyLen = strlen(key);

    printf("\nEncrypted Text: ");

    for (i = 0; text[i] != '\0'; i++)
    {
        if (isalpha(text[i]))
        {
            char ch = toupper(text[i]);
            char k = toupper(key[j % keyLen]);

            char encrypted = ((ch - 'A' + k - 'A') % 26) + 'A';

            printf("%c", encrypted);
            j++;
        }
        else
        {
            printf("%c", text[i]);
        }
    }

    printf("\n");

    return 0;
}
