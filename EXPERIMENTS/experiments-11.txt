
#include <stdio.h>
#include <math.h>

int main()
{
    long double factorial = 1.0;
    long double uniqueKeys;
    int i;

    // Calculate 25!
    for (i = 1; i <= 25; i++)
    {
        factorial *= i;
    }

    // Calculate 5! * 5!
    long double equivalent = 1.0;

    for (i = 1; i <= 5; i++)
    {
        equivalent *= i;
    }

    equivalent = equivalent * equivalent;

    // Calculate effectively unique keys
    uniqueKeys = factorial / equivalent;

    printf("Total possible keys = %.0Lf\n", factorial);

    printf("Approximate power of 2 = 2^%.2Lf\n",
           log2l(factorial));

    printf("\nEffectively unique keys = %.0Lf\n",
           uniqueKeys);

    printf("Approximate power of 2 = 2^%.2Lf\n",
           log2l(uniqueKeys));

    return 0;
}
