
#include <stdio.h>
#include <string.h>
#include <ctype.h>

char matrix[5][5] = {
    {'M','F','H','I','K'},
    {'U','N','O','P','Q'},
    {'Z','V','W','X','Y'},
    {'E','L','A','R','G'},
    {'D','S','T','B','C'}
};

void findPosition(char ch, int *row, int *col)
{
    int i, j;

    if (ch == 'J')
        ch = 'I';

    for (i = 0; i < 5; i++)
    {
        for (j = 0; j < 5; j++)
        {
            if (matrix[i][j] == ch)
            {
                *row = i;
                *col = j;
                return;
            }
        }
    }
}

void encrypt(char a, char b)
{
    int r1, c1, r2, c2;

    findPosition(a, &r1, &c1);
    findPosition(b, &r2, &c2);

    if (r1 == r2)
    {
        printf("%c%c", matrix[r1][(c1 + 1) % 5],
                       matrix[r2][(c2 + 1) % 5]);
    }
    else if (c1 == c2)
    {
        printf("%c%c", matrix[(r1 + 1) % 5][c1],
                       matrix[(r2 + 1) % 5][c2]);
    }
    else
    {
        printf("%c%c", matrix[r1][c2],
                       matrix[r2][c1]);
    }
}

int main()
{
    char text[200], msg[200];
    int i, j = 0, len;

    printf("Enter plaintext: ");
    fgets(text, sizeof(text), stdin);

    // Remove spaces and special characters
    for (i = 0; text[i] != '\0'; i++)
    {
        if (isalpha((unsigned char)text[i]))
        {
            msg[j++] = toupper((unsigned char)text[i]);
        }
    }

    msg[j] = '\0';

    // Replace J with I
    for (i = 0; i < j; i++)
    {
        if (msg[i] == 'J')
            msg[i] = 'I';
    }

    // Prepare letter pairs
    len = 0;

    for (i = 0; i < j; )
    {
        char a = msg[i++];
        char b;

        if (i < j && msg[i] != a)
            b = msg[i++];
        else
            b = 'X';

        msg[len++] = a;
        msg[len++] = b;
    }

    msg[len] = '\0';

    printf("Encrypted text: ");

    for (i = 0; i < len; i += 2)
    {
        encrypt(msg[i], msg[i + 1]);
        printf(" ");
    }

    return 0;
}
