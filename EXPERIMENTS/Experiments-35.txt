
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

int main()
{
    char plaintext[100], ciphertext[100];
    int key[100], i, len;

    printf("Enter plaintext (A-Z): ");
    scanf("%99s", plaintext);

    len = strlen(plaintext);

    srand(time(NULL));

    // Generate random key
    printf("\nRandom Key: ");
    for(i = 0; i < len; i++)
    {
        key[i] = rand() % 26;
        printf("%d ", key[i]);
    }

    // Encryption
    for(i = 0; i < len; i++)
    {
        ciphertext[i] = ((plaintext[i] - 'A' + key[i]) % 26) + 'A';
    }
    ciphertext[i] = '\0';

    printf("\n\nEncrypted Text: %s", ciphertext);

    // Decryption
    printf("\nDecrypted Text: ");
    for(i = 0; i < len; i++)
    {
        printf("%c", ((ciphertext[i] - 'A' - key[i] + 26) % 26) + 'A');
    }

    printf("\n");

    return 0;
}
