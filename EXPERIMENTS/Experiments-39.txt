
#include <stdio.h>
#include <ctype.h>
#include <string.h>

int main() {
    char cipher[1000];

    printf("Enter ciphertext:\n");
    fgets(cipher, sizeof(cipher), stdin);

    printf("\nPossible plaintexts:\n");

    for (int key = 0; key < 26; key++) {
        printf("\nKey %d: ", key);

        for (int i = 0; cipher[i] != '\0'; i++) {
            if (isalpha(cipher[i])) {
                int c = toupper(cipher[i]) - 'A';
                int p = (c - key + 26) % 26;

                char ch = p + 'A';

                if (islower(cipher[i]))
                    ch = tolower(ch);

                putchar(ch);
            }
            else {
                putchar(cipher[i]);
            }
        }
    }

    return 0;
}
