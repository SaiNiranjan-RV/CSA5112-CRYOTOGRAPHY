
#include <stdio.h>

int main()
{
    int K[2][2] = {
        {9, 4},
        {5, 7}
    };

    int P[2][2] = {
        {1, 0},
        {0, 1}
    };

    int C[2][2] = {0};
    int i, j, k;

    printf("Original Key Matrix:\n");

    for (i = 0; i < 2; i++)
    {
        for (j = 0; j < 2; j++)
            printf("%d ", K[i][j]);

        printf("\n");
    }

    // Encryption of chosen plaintext
    for (i = 0; i < 2; i++)
    {
        for (j = 0; j < 2; j++)
        {
            for (k = 0; k < 2; k++)
            {
                C[i][j] += K[i][k] * P[k][j];
            }

            C[i][j] %= 26;
        }
    }

    printf("\nChosen Plaintext Matrix:\n");

    for (i = 0; i < 2; i++)
    {
        for (j = 0; j < 2; j++)
            printf("%d ", P[i][j]);

        printf("\n");
    }

    printf("\nCorresponding Ciphertext Matrix:\n");

    for (i = 0; i < 2; i++)
    {
        for (j = 0; j < 2; j++)
            printf("%d ", C[i][j]);

        printf("\n");
    }

    // Since P is an identity matrix:
    // K = C * inverse(P) = C

    printf("\nRecovered Key Matrix:\n");

    for (i = 0; i < 2; i++)
    {
        for (j = 0; j < 2; j++)
            printf("%d ", C[i][j]);

        printf("\n");
    }

    return 0;
}
